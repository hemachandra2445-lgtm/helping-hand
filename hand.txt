        function redirectToUPI(provider) {
            const amount = document.getElementById('customAmount').value || '100';
            const note = encodeURIComponent(`SVEC Relief Fund for ${currentStudent}`);
            const upiString = `upi://pay?pa=${targetUPI}&pn=${encodeURIComponent(payeeName)}&am=${amount}&cu=INR&tn=${note}`;

            if (provider === 'gpay') {
                window.location.href = `intent://pay?pa=${targetUPI}&pn=${encodeURIComponent(payeeName)}&am=${amount}&cu=INR&tn=${note}#Intent;scheme=upi;package=com.google.android.apps.nfc.plugin.card.bpay;end`;
            } else if (provider === 'phonepe') {
                window.location.href = `intent://pay?pa=${targetUPI}&pn=${encodeURIComponent(payeeName)}&am=${amount}&cu=INR&tn=${note}#Intent;scheme=upi;package=com.phonepe.app;end`;
            } else if (provider === 'paytm') {
                window.location.href = `intent://pay?pa=${targetUPI}&pn=${encodeURIComponent(payeeName)}&am=${amount}&cu=INR&tn=${note}#Intent;scheme=upi;package=net.one97.paytm;end`;
            } else {
                window.location.href = upiString;
            }
        }

        document.getElementById('emergencyRequestForm').addEventListener('submit', async function(event) {
            event.preventDefault();
            const submitBtn = document.getElementById('submitBtn');
            const originalBtnContent = submitBtn.innerHTML;

            submitBtn.disabled = true;
            submitBtn.innerHTML = `<i class="fa-solid fa-spinner fa-spin"></i> Submitting...`;

            try {
                const formData = new FormData(this);
                
                // Use uploaded photo or fallback
                const studentPhoto = currentUploadedPhotoBase64 || 'https://images.unsplash.com/photo-1534528741775-53994a69daeb?w=600&auto=format&fit=crop&q=80';

                const newReq = {
                    id: Date.now(),
                    name: formData.get('Student Name') || 'New Student',
                    roll: formData.get('Roll Number') || 'N/A',
                    dept: `${formData.get('Department')} (${formData.get('Year of Study')})`,
                    hospital: formData.get('Hospital Name') || 'Hospital',
                    amount: `₹${Number(formData.get('Requested Amount (INR)')).toLocaleString('en-IN')}`,
                    category: formData.get('Emergency Category') || 'Medical Emergency',
                    phone: formData.get('Phone Number') || 'N/A',
                    status: 'Pending Verification',
                    desc: formData.get('Emergency Description') || 'Submitted emergency request pending review.',
                    photo: studentPhoto
                };
                fundRequests.unshift(newReq);
                saveState();

                const response = await fetch('https://api.web3forms.com/submit', {
                    method: 'POST',
                    body: formData
                });

                const data = await response.json();

                if (data.success) {
                    alert("Your emergency request with student photograph has been submitted for admin verification!");
                    this.reset();
                    document.getElementById('studentPhotoPreview').classList.add('hidden');
                    document.getElementById('previewPlaceholder').classList.remove('hidden');
                    currentUploadedPhotoBase64 = '';
                    showView('homeView');
                } else {
                    alert("Submission error: " + data.message);
                }
            } catch (error) {
                alert("Request logged with photo. Details forwarded for administrative verification.");
                this.reset();
                document.getElementById('studentPhotoPreview').classList.add('hidden');
                document.getElementById('previewPlaceholder').classList.remove('hidden');
                currentUploadedPhotoBase64 = '';
                showView('homeView');
            } finally {
                submitBtn.disabled = false;
                submitBtn.innerHTML = originalBtnContent;
            }
        });
